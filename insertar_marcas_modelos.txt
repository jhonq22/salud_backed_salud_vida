6. QUERYS DE VERIFICACIÓN POST-EJECUCIÓN
================================================================================

-- Verificar las 5 marcas activas por cada sección:
SELECT id, marca, tipo_marca_id, estatus FROM marcas WHERE estatus = 1 ORDER BY tipo_marca_id, id;

-- Verificar cantidad de modelos activos por cada marca y categoría:
SELECT 
  ma.tipo_marca_id,
  CASE ma.tipo_marca_id
    WHEN 1 THEN 'GENERADOR'
    WHEN 2 THEN 'ELECTRODO VENTRICULAR'
    WHEN 3 THEN 'ELECTRODO AURICULAR'
  END as categoria,
  ma.id as marca_id,
  ma.marca,
  COUNT(mo.id) as total_modelos_activos
FROM marcas ma
JOIN modelos mo ON mo.marca_id = ma.id AND mo.estatus = 1
WHERE ma.estatus = 1
GROUP BY ma.tipo_marca_id, ma.id, ma.marca
ORDER BY ma.tipo_marca_id, ma.marca;
